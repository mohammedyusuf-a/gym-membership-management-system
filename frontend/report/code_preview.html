js/data.js — GymStore LocalStorage State Engine & Permanent User Data 
// Unified Gym State Manager with Permanent LocalStorage Persistence
class GymStore {
  constructor() {
    this.init();
  }

  // Persistent User Profile & Permanent DP Storage
  updateUserProfile(emailOrId, data) {
    const users = this.getUsers();
    const clean = (emailOrId || '').trim().toLowerCase();
    const index = users.findIndex(u => (u.email && u.email.toLowerCase() === clean) || u.id === emailOrId);
    
    if (index !== -1) {
      users[index] = { ...users[index], ...data };
      localStorage.setItem(STORAGE_KEYS.USERS, JSON.stringify(users));

      // Permanent dedicated key ensuring uploaded DP never vanishes
      if (data.dpImage) {
        localStorage.setItem('FITNESS_CLUB_USER_DP_' + clean, data.dpImage);
        localStorage.setItem('FITNESS_CLUB_ACTIVE_DP', data.dpImage);
      }

      const currentAuth = this.getAdminAuth();
      if (currentAuth && (currentAuth.email.toLowerCase() === clean || currentAuth.id === emailOrId)) {
        const updatedAuth = { ...currentAuth, name: users[index].name, dpImage: users[index].dpImage };
        localStorage.setItem(STORAGE_KEYS.AUTH, JSON.stringify(updatedAuth));
      }
      return { success: true, user: users[index] };
    }
  }

  // Real-Time Member Fast Check-in Processing
  checkInMember(query) {
    const members = this.getMembers();
    const target = members.find(m => m.fullName.toLowerCase().includes(query.toLowerCase()) || m.id === query);
    if (!target) return { success: false, message: 'Member not found.' };
    if (target.status === 'Expired') return { success: false, message: 'Membership expired! Please renew.', member: target };
    
    this.addAttendanceLog({ memberId: target.id, memberName: target.fullName, checkInTime: new Date().toLocaleTimeString(), status: 'Checked In' });
    return { success: true, message: `Check-in recorded: ${target.fullName}`, member: target };
  }
}
js/app.js — Application Controller, View Routing & DP File Reader 
const App = {
  // Navigation View Router
  navigateTo(viewName) {
    document.querySelectorAll('.view-panel').forEach(p => p.classList.remove('active'));
    document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
    
    const target = document.getElementById(`view-${viewName}`);
    if (target) {
      target.classList.add('active');
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }
    if (viewName === 'profile') this.populateCustomerProfileUI();
  },

  // Permanent Profile Picture (DP) Upload with FileReader base64
  handleProfileDPUpload(event) {
    const file = event.target.files && event.target.files[0];
    if (!file || !file.type.startsWith('image/')) return;
    if (file.size > 3 * 1024 * 1024) return this.showToast('Max size 3MB', 'error');

    const reader = new FileReader();
    reader.onload = (e) => {
      const base64 = e.target.result;
      const auth = window.GymDB.getAdminAuth();
      const email = (auth.email || 'yusuf@fitnessathlete.com').toLowerCase();

      // Permanent persistence across sessions, reloads & logins
      localStorage.setItem('FITNESS_CLUB_USER_DP_' + email, base64);
      localStorage.setItem('FITNESS_CLUB_ACTIVE_DP', base64);
      window.GymDB.updateUserProfile(email, { dpImage: base64 });

      this.updateUserSessionUI({ ...auth, dpImage: base64 });
      this.populateCustomerProfileUI();
      this.showToast('Profile photo saved permanently!', 'success');
    };
    reader.readAsDataURL(file);
  },

  // Creator & Owner Master Admin Key Portal Modal
  openCreatorAdminModal() {
    this.closeAuthGateway();
    document.getElementById('admin-pass-input').value = '';
    this.openModal('modal-admin-portal');
  }
};
